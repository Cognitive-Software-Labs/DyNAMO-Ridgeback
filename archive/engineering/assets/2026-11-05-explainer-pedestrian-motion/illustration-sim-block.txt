  const path = keys => t => {
    if (t <= keys[0][0]) return [keys[0][1], keys[0][2]];
    for (let i = 1; i < keys.length; i++) { const a = keys[i - 1], b = keys[i]; if (t <= b[0]) { const u = (t - a[0]) / (b[0] - a[0]); return [a[1] + u * (b[1] - a[1]), a[2] + u * (b[2] - a[2])]; } }
    const l = keys[keys.length - 1]; return [l[1], l[2]];
  };

  const segDist = (p, a, b) => { const ab = sub(b, a), ap = sub(p, a); const s = Math.max(0, Math.min(1, (ap[0] * ab[0] + ap[1] * ab[1]) / (ab[0] ** 2 + ab[1] ** 2))); return [len(sub(p, [a[0] + s * ab[0], a[1] + s * ab[1]])), s]; };

  const frames = [];
  let R = [2.2, 4.6], state = 'LOCKED', hidden = 0, seen = 0, sinceLock = 99, heading = 0, lastSeen = leaderAt(0);
  for (let i = 0; i <= N; i++) {
    const t = i * DT, L = leaderAt(t), P = others.map(o => o.at(t));
    const blocked = P.some(p => { const [d, s] = segDist(p, R, L); return d < 0.32 && s > 0.05 && s < 0.95; });
    let why;
    if (blocked) {
      seen = 0; hidden += DT;
      if (state === 'LOCKED') state = 'UNCERTAIN';
      if (state === 'UNCERTAIN' && hidden > 1.5) state = 'LOST';
      why = state === 'LOST' ? 'lost' : 'hidden';
    } else {
      hidden = 0;
      if (state !== 'LOCKED') { seen += DT; why = 'confirming'; if (seen >= 0.6) { state = 'LOCKED'; sinceLock = 0; seen = 0; } }
    }
    if (state === 'LOCKED') { lastSeen = L; sinceLock += DT; why = sinceLock < 1.6 ? 'recovered' : 'locked'; }
    // motion: follow when locked, keep clear of everyone always
    let v = [0, 0];
    if (state === 'LOCKED') {
      const d = sub(L, R), dist = len(d), tgt = [L[0] - 1.4 * d[0] / dist, L[1] - 1.4 * d[1] / dist];
