cmake_minimum_required(VERSION 3.16)
project(DirectFps)
find_package(gz-gui8 REQUIRED)
find_package(gz-plugin2 REQUIRED)
find_package(Qt5 REQUIRED COMPONENTS Core Quick)
set(CMAKE_AUTOMOC ON)
cmake_policy(SET CMP0100 NEW)
qt5_add_resources(RSC DirectFps.qrc)
add_library(DirectFps SHARED DirectFps.cc DirectFps.hh ${RSC})
target_link_libraries(DirectFps gz-gui8::gz-gui8 gz-plugin2::gz-plugin2 Qt5::Core Qt5::Quick)
