cmake_minimum_required(VERSION 3.16)
project(leader_follow_leg_bridge)
find_package(ament_cmake REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(tf2_geometry_msgs REQUIRED)
find_package(OpenCV REQUIRED)
if(NOT LEG_SOURCE)
  message(FATAL_ERROR "Set LEG_SOURCE to the prepared upstream leg_detector package")
endif()
add_executable(leg_cli leg_cli.cpp ${LEG_SOURCE}/src/laser_processor.cpp ${LEG_SOURCE}/src/cluster_features.cpp)
target_compile_features(leg_cli PRIVATE cxx_std_17)
target_include_directories(leg_cli PRIVATE ${LEG_SOURCE}/include ${OpenCV_INCLUDE_DIRS})
ament_target_dependencies(leg_cli sensor_msgs tf2_geometry_msgs)
target_link_libraries(leg_cli ${OpenCV_LIBS})
